**Queries (src/queries/recovery.ts):**
- `recoverySessionsOptions(userId: string)` — fetches workout_sessions from last 42 days with `started_at` and `total_volume`. Uses Zod-transformed data to avoid weight doubling.
- `wearableRecoveryOptions(userId: string)` — fetches from `external_activities` where provider IN ('garmin', 'fitbit'), extracts `raw_data` JSONB for sleep/HRV if available. Returns null gracefully if no data.
- `activeCyclePositionOptions(userId: string)` — fetches the user's active training cycle from `training_cycles` where `status = 'active'`, returning `{ current_week, duration_weeks, status }`. Uses existing `cycleListOptions` pattern from `src/queries/cycles.ts`. Returns null if no active cycle. Query: `supabase.from('training_cycles').select('current_week, duration_weeks, status').eq('user_id', userId).eq('status', 'active').maybeSingle()`.

**Hook (src/hooks/useRecoveryScore.ts):**
```typescript
export function useRecoveryScore() {
  const { user } = useAuth();
  const { data: sessions } = useQuery(recoverySessionsOptions(user?.id ?? ""));
  const { data: wearable } = useQuery(wearableRecoveryOptions(user?.id ?? ""));
  const { data: activeCycle } = useQuery(activeCyclePositionOptions(user?.id ?? ""));

  const result = useMemo(() => {
    if (!sessions) return null;
    // Compute daysSinceFirstSession from oldest session
    const cyclePosition = activeCycle ? {
      currentWeek: activeCycle.current_week,
      durationWeeks: activeCycle.duration_weeks,
      status: activeCycle.status as "active" | "completed" | "draft",
    } : null;
    return computeReadinessScore({ sessions: transformedSessions, daysSinceFirstSession, cyclePosition });
  }, [sessions, activeCycle]);

  return { recovery: result, wearable, isLoading: !sessions };
}
```
