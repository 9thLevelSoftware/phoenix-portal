**Hook (src/hooks/useOnboarding.ts):**
Implements the detection logic from research Pitfall 4:
```typescript
export function useOnboarding() {
  const { user } = useAuth();
  const queryClient = useQueryClient();

  const { data: onboarding, isPending: onboardingLoading } = useQuery({
    ...onboardingOptions(user?.id ?? ""),
    enabled: !!user,
  });

  const { data: workoutCount, isPending: workoutsLoading } = useQuery({
    ...hasWorkoutsOptions(user?.id ?? ""),
    enabled: !!user && onboarding === null, // Only check if no onboarding row
  });

  const isPending = onboardingLoading || (onboarding === null && workoutsLoading);
  const hasExistingData = (workoutCount ?? 0) > 0;

  // Decision tree:
  // 1. No onboarding row + no workouts → needsOnboarding (new user)
  // 2. No onboarding row + has workouts → needsWhatsNew (v1.0 mobile user)
  //    Also: auto-create onboarding row with completed_at=now() so they don't see full onboarding
  // 3. Has row + version_seen < '1.1' + !dismissed_whats_new → needsWhatsNew
  // 4. Has row + version_seen >= '1.1' → nothing needed
  const needsOnboarding = !isPending && onboarding === null && !hasExistingData;
  const needsWhatsNew = !isPending && (
    (onboarding === null && hasExistingData) ||
    (onboarding && (!onboarding.version_seen || onboarding.version_seen < '1.1') && !onboarding.dismissed_whats_new)
  );

  // Mutation to complete onboarding
  const completeOnboarding = useMutation(/* upsert user_onboarding with completed_at, version_seen='1.1' */);
  const dismissWhatsNew = useMutation(/* update dismissed_whats_new=true */);
  const dismissHint = useMutation(/* update dismissed_hints JSONB */);

  return { needsOnboarding, needsWhatsNew, onboarding, isPending, completeOnboarding, dismissWhatsNew, dismissHint };
}
```

export function AppLayout() {
  useRealtimeSync();
  const { needsOnboarding, needsWhatsNew, completeOnboarding, dismissWhatsNew } = useOnboarding();

      <Toaster />

  );
}
